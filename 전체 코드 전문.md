bool UWeaponCarryHandler::AttachWeaponToHand()
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	const bool bAttached = WeaponData
		&& Component->AttachWeaponToCharacterSocket(
			WeaponData->AttachSocketName,
			WeaponData->EquippedMeshTransform);

	if (bAttached)
	{
		State.Carry.bWeaponInHand = true;
	}

	return bAttached;
}

bool UWeaponCarryHandler::AttachWeaponToHolster()
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	const bool bAttached = WeaponData
		&& Component->AttachWeaponToCharacterSocket(
			WeaponData->HolsterSocketName,
			WeaponData->HolsteredMeshTransform);

	if (bAttached)
	{
		State.Carry.bWeaponInHand = false;
	}

	return bAttached;
}

void UWeaponCarryHandler::SetWeaponCarryState(EWeaponCarryState NewState)
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (State.Carry.WeaponCarryState == NewState)
	{
		return;
	}

	const EWeaponCarryState PreviousState = State.Carry.WeaponCarryState;
	State.Carry.WeaponCarryState = NewState;
	Component->OnWeaponCarryStateChanged.Broadcast(PreviousState, State.Carry.WeaponCarryState);
}

bool UWeaponCarryHandler::EnterHolsteredCarryState()
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	StopAutoHolsterTimer();
	Component->StopContinuousAttack();
	Component->ClearAimSolution();

	const bool bAttached = AttachWeaponToHolster();
	if (bAttached)
	{
		SetWeaponCarryState(EWeaponCarryState::Holstered);

		// Holstering 중 들어온 Fire/Reload/Aim 요청은 수납이 끝난 뒤 Draw로 이어집니다.
		if (State.Carry.PendingWeaponAction != EPendingWeaponAction::None)
		{
			Component->BroadcastWeaponCommand(EWeaponCommand::Draw);
		}
	}

	return bAttached;
}

{

	StopAutoHolsterTimer();
	SetWeaponCarryState(EWeaponCarryState::Drawing);
}

bool UWeaponCarryHandler::EnterReadyCarryState()
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	const bool bAttached = AttachWeaponToHand();
	if (bAttached)
	{
		SetWeaponCarryState(EWeaponCarryState::Ready);
		StartAutoHolsterTimer();

		if (State.Carry.PendingWeaponAction != EPendingWeaponAction::None)
		{
			const EPendingWeaponAction ReadyAction = State.Carry.PendingWeaponAction;
			State.Carry.PendingWeaponAction = EPendingWeaponAction::None;
			Component->OnPendingWeaponActionReady.Broadcast(ReadyAction);
		}
	}

	return bAttached;
}

	StopAutoHolsterTimer();
	SetWeaponCarryState(EWeaponCarryState::Holstering);
}

		Component,

	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!State.Carry.bDrawInProgress || !State.Carry.bWaitingForDrawAlignment || State.Carry.WeaponCarryState != EWeaponCarryState::Drawing)
	{
		return false;
	}

	State.Carry.bWaitingForDrawAlignment = false;
	return StartDrawPresentation();
}

bool UWeaponCarryHandler::StartDrawPresentation()
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!State.Carry.bDrawInProgress || State.Carry.bWaitingForDrawAlignment || !Context.LinkedAnimationMesh || !Component->HasEquippedWeapon())
	{
		FinishDrawWeapon(false);
		return false;
	}

	UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData)
	{
		FinishDrawWeapon(false);
		return false;
	}

	UAnimMontage* DrawMontage = WeaponData->EquipMontage;

	// Equip Montage가 없는 무기는 손에 먼저 부착한 뒤 Ready 전환 시점까지 기다립니다.
	if (!DrawMontage)
	{
		State.Carry.bAttachedToHandDuringDraw = AttachWeaponToHand();
		if (!State.Carry.bAttachedToHandDuringDraw)
		{
			FinishDrawWeapon(false);
			return false;
		}

		const float ReadyDelay = FMath::Max(0.f, WeaponData->DrawReadyDelay);
		if (ReadyDelay <= 0.f)
		{
			FinishDrawWeapon(true);
			return true;
		}

		UWorld* World = Context.World.Get();
		if (!World)
		{
			FinishDrawWeapon(false);
			return false;
		}

		World->GetTimerManager().SetTimer(State.Carry.DrawReadyTimerHandle, Component, &UWeaponComponent::HandleDrawReadyTimerElapsed, ReadyDelay, false);
		return true;
	}

	UAnimInstance* AnimInstance = Context.LinkedAnimationMesh->GetAnimInstance();
	if (!AnimInstance)
	{
		FinishDrawWeapon(false);
		return false;
	}

	State.Carry.ActiveDrawAnimInstance = AnimInstance;
	State.Carry.ActiveDrawMontage = DrawMontage;

	AnimInstance->OnPlayMontageNotifyBegin.AddUniqueDynamic(Component, &UWeaponComponent::HandleDrawMontageNotify);

	const float PlayedDuration = AnimInstance->Montage_Play(DrawMontage, 1.f);
	if (PlayedDuration <= 0.f)
	{
		AnimInstance->OnPlayMontageNotifyBegin.RemoveDynamic(Component, &UWeaponComponent::HandleDrawMontageNotify);
		FinishDrawWeapon(false);
		return false;
	}

	FOnMontageEnded EndDelegate;
	EndDelegate.BindUObject(Component, &UWeaponComponent::HandleDrawMontageEnded);
	AnimInstance->Montage_SetEndDelegate(EndDelegate, DrawMontage);
	return true;
}

void UWeaponCarryHandler::HandleDrawReadyTimerElapsed()
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	const bool bSucceeded = State.Carry.bDrawInProgress && !State.Carry.bWaitingForDrawAlignment 
		&& State.Carry.WeaponCarryState == EWeaponCarryState::Drawing && State.Carry.bAttachedToHandDuringDraw;

	FinishDrawWeapon(bSucceeded);
}

void UWeaponCarryHandler::HandleDrawMontageNotify(FName NotifyName, const FBranchingPointNotifyPayload& BranchingPointPayload)
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	const FAnimMontageInstance* MontageInstance = State.Carry.ActiveDrawAnimInstance ? State.Carry.ActiveDrawAnimInstance->GetActiveInstanceForMontage(State.Carry.ActiveDrawMontage) : nullptr;
	if (!MontageInstance || MontageInstance->GetInstanceID() != BranchingPointPayload.MontageInstanceID)
	{
		return;
	}

	if (!State.Carry.bDrawInProgress || NotifyName != Component->AttachToHandNotifyName || !Context.LinkedAnimationMesh || BranchingPointPayload.SkelMeshComponent != Context.LinkedAnimationMesh)
	{
		return;
	}

	State.Carry.bAttachedToHandDuringDraw = AttachWeaponToHand();
}

void UWeaponCarryHandler::HandleDrawMontageEnded(UAnimMontage* Montage, bool bInterrupted)
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!State.Carry.bDrawInProgress || Montage != State.Carry.ActiveDrawMontage)
	{
		return;
	}

	if (State.Carry.ActiveDrawAnimInstance)
	{
		State.Carry.ActiveDrawAnimInstance->OnPlayMontageNotifyBegin.RemoveDynamic(Component, &UWeaponComponent::HandleDrawMontageNotify);
	}

	bool bSucceeded = !bInterrupted;
	if (bSucceeded && !State.Carry.bAttachedToHandDuringDraw)
	{
		UE_LOG(LogTemp, Warning,
			TEXT("%s: Draw montage ended without '%s'. Applying hand attachment as a fallback."),
			*GetNameSafe(Component),
			*Component->AttachToHandNotifyName.ToString());
		bSucceeded = AttachWeaponToHand();
	}

	FinishDrawWeapon(bSucceeded);
}

void UWeaponCarryHandler::FinishDrawWeapon(bool bSucceeded)
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (UWorld* World = Context.World.Get())
	{
		World->GetTimerManager().ClearTimer(State.Carry.DrawReadyTimerHandle);
	}

	State.Carry.bDrawInProgress = false;
	State.Carry.bWaitingForDrawAlignment = false;
	State.Carry.bLastDrawSucceeded = bSucceeded;
	State.Carry.bAttachedToHandDuringDraw = false;
	State.Carry.ActiveDrawAnimInstance = nullptr;
	State.Carry.ActiveDrawMontage = nullptr;

	if (!bSucceeded)
	{
		State.Carry.PendingWeaponAction = EPendingWeaponAction::None;
	}

	Component->OnWeaponDrawFinished.Broadcast(bSucceeded);
}

void UWeaponCarryHandler::CancelDrawWeapon(float BlendOutTime)
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!State.Carry.bDrawInProgress)
	{
		return;
	}

	if (State.Carry.ActiveDrawAnimInstance && State.Carry.ActiveDrawMontage && State.Carry.ActiveDrawAnimInstance->Montage_IsPlaying(State.Carry.ActiveDrawMontage))
	{
		State.Carry.ActiveDrawAnimInstance->Montage_Stop(FMath::Max(0.f, BlendOutTime), State.Carry.ActiveDrawMontage);
		return;
	}

	if (State.Carry.ActiveDrawAnimInstance)
	{
		State.Carry.ActiveDrawAnimInstance->OnPlayMontageNotifyBegin.RemoveDynamic(Component, &UWeaponComponent::HandleDrawMontageNotify);
	}

	FinishDrawWeapon(false);
}

bool UWeaponCarryHandler::ExecuteHolsterWeapon()
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (State.Carry.bHolsterInProgress || !Context.LinkedAnimationMesh || !Component->HasEquippedWeapon())
	{
		return false;
	}

	EnterHolsteringCarryState();
	State.Carry.bLastHolsterSucceeded = false;
	State.Carry.bAttachedToHolsterDuringAnimation = false;

	UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	UAnimMontage* HolsterMontage = WeaponData ? WeaponData->HolsterMontage : nullptr;

	UAnimInstance* AnimInstance = Context.LinkedAnimationMesh->GetAnimInstance();

	if (!HolsterMontage)
	{
		const bool bAttached = AttachWeaponToHolster();
		State.Carry.bLastHolsterSucceeded = bAttached;
		Component->OnWeaponHolsterFinished.Broadcast(bAttached);
		return bAttached;
	}

	if (!AnimInstance)
	{
		Component->OnWeaponHolsterFinished.Broadcast(false);
		return false;
	}

	State.Carry.ActiveHolsterAnimInstance = AnimInstance;
	State.Carry.ActiveHolsterMontage = HolsterMontage;
	State.Carry.bHolsterInProgress = true;

	AnimInstance->OnPlayMontageNotifyBegin.AddUniqueDynamic(Component, &UWeaponComponent::HandleHolsterMontageNotify);

	const float PlayedDuration = AnimInstance->Montage_Play(HolsterMontage, 1.f, EMontagePlayReturnType::MontageLength, 0.f, true);
	if (PlayedDuration <= 0.f)
	{
		AnimInstance->OnPlayMontageNotifyBegin.RemoveDynamic(Component, &UWeaponComponent::HandleHolsterMontageNotify);
		FinishHolsterWeapon(false);
		return false;
	}

	FOnMontageEnded EndDelegate;
	EndDelegate.BindUObject(Component, &UWeaponComponent::HandleHolsterMontageEnded);
	AnimInstance->Montage_SetEndDelegate(EndDelegate, HolsterMontage);
	return true;
}

bool UWeaponAimHandler::UpdateWeaponAimSolution(FVector ViewLocation, FVector ViewDirection, TEnumAsByte<ECollisionChannel> TraceChannel, FLinearColor DebugColor)
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!RequiresAimSolutionUpdate())
	{
		ClearAimSolution();
		return false;
	}

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	UWorld* World = Context.World.Get();
	const FVector NormalizedViewDirection = ViewDirection.GetSafeNormal();

	if (!WeaponData || !World || WeaponData->EffectiveRange <= 0.f || NormalizedViewDirection.IsNearlyZero())
	{
		ClearAimSolution();
		return false;
	}

	const FVector TraceEnd = ViewLocation + NormalizedViewDirection * WeaponData->EffectiveRange;

	FCollisionQueryParams QueryParams(SCENE_QUERY_STAT(WeaponAimSolution), false);
	QueryParams.AddIgnoredComponent(Component);

	if (Context.LinkedAnimationMesh)
	{
		QueryParams.AddIgnoredComponent(Context.LinkedAnimationMesh.Get());
	}

	if (Context.LinkedBodyCollisionComponent)
	{
		QueryParams.AddIgnoredComponent(Context.LinkedBodyCollisionComponent.Get());
	}

	FHitResult HitResult;
	const bool bBlockingHit = World->LineTraceSingleByChannel(HitResult, ViewLocation, TraceEnd, TraceChannel, QueryParams);
	const FVector AimTargetLocation = bBlockingHit ? HitResult.ImpactPoint : TraceEnd;
	const FVector MuzzleLocation = Component->GetMuzzleTransform().GetLocation();
	const FVector ShotDirection = ResolveShotDirection(MuzzleLocation, AimTargetLocation);

	if (ShotDirection.IsNearlyZero())
	{
		ClearAimSolution();
		return false;
	}

	State.Aim.CurrentAimSolution.bValid = true;
	State.Aim.CurrentAimSolution.bBlockingHit = bBlockingHit;
	State.Aim.CurrentAimSolution.ViewLocation = ViewLocation;
	State.Aim.CurrentAimSolution.MuzzleLocation = MuzzleLocation;
	State.Aim.CurrentAimSolution.AimTargetLocation = AimTargetLocation;
	State.Aim.CurrentAimSolution.ShotDirection = ShotDirection;

	if (DebugColor.A > 0.f)
	{
		FColor DebugDrawColor = DebugColor.ToFColor(true);
		DebugDrawColor.A = 255;

		DrawDebugDirectionalArrow(World, ViewLocation, AimTargetLocation, 10.f, DebugDrawColor, false, 0.f, 0, 1.f);
		DrawDebugPoint(World, AimTargetLocation, 10.f, DebugDrawColor, false, 0.f, 0);
		DrawDebugString(World, AimTargetLocation + FVector(0.f, 0.f, 20.f), TEXT("Aim Target"), nullptr, DebugDrawColor, 0.f);
	}

	return true;
}

void UWeaponAimHandler::UpdateWeaponVisualAimCorrection(float DeltaSeconds)
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();

	if (!WeaponData || !State.Aim.CurrentAimSolution.bValid)
	{
		ResetWeaponVisualAimCorrection(DeltaSeconds);
		return;
	}

	const FTransform MuzzleTransform = Component->GetMuzzleTransform();
	const FVector MuzzleLocation = MuzzleTransform.GetLocation();
	const FVector CurrentMuzzleDirection = MuzzleTransform.GetRotation().GetForwardVector().GetSafeNormal();
	const FVector DesiredMuzzleDirection = (State.Aim.CurrentAimSolution.AimTargetLocation - MuzzleLocation).GetSafeNormal();

	if (CurrentMuzzleDirection.IsNearlyZero() || DesiredMuzzleDirection.IsNearlyZero())
	{
		ResetWeaponVisualAimCorrection(DeltaSeconds);
		return;
	}

	const FRotator CurrentMuzzleRotation = CurrentMuzzleDirection.Rotation();
	const FRotator DesiredMuzzleRotation = DesiredMuzzleDirection.Rotation();
	const FRotator DeltaRotation = (DesiredMuzzleRotation - CurrentMuzzleRotation).GetNormalized();

	State.Aim.TargetVisualAimCorrectionPitch = FMath::Clamp(State.Aim.VisualAimCorrectionPitch + DeltaRotation.Pitch, 
		-WeaponData->MaxVisualAimCorrectionPitch, WeaponData->MaxVisualAimCorrectionPitch);

	State.Aim.TargetVisualAimCorrectionYaw = FMath::Clamp(State.Aim.VisualAimCorrectionYaw + DeltaRotation.Yaw,
		-WeaponData->MaxVisualAimCorrectionYaw, WeaponData->MaxVisualAimCorrectionYaw);

	const float SafeDeltaSeconds = FMath::Max(DeltaSeconds, 0.f);

	State.Aim.VisualAimCorrectionPitch = FMath::FInterpTo(State.Aim.VisualAimCorrectionPitch, State.Aim.TargetVisualAimCorrectionPitch,
		SafeDeltaSeconds, WeaponData->VisualAimCorrectionInterpSpeed);

	State.Aim.VisualAimCorrectionYaw = FMath::FInterpTo(State.Aim.VisualAimCorrectionYaw, State.Aim.TargetVisualAimCorrectionYaw,
		SafeDeltaSeconds, WeaponData->VisualAimCorrectionInterpSpeed);
}

void UWeaponAimHandler::ResetWeaponVisualAimCorrection(float DeltaSeconds)
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	const float InterpSpeed = WeaponData ? WeaponData->VisualAimCorrectionInterpSpeed : 10.f;
	const float SafeDeltaSeconds = FMath::Max(DeltaSeconds, 0.f);

	State.Aim.TargetVisualAimCorrectionPitch = 0.f;
	State.Aim.TargetVisualAimCorrectionYaw = 0.f;

	State.Aim.VisualAimCorrectionPitch = FMath::FInterpTo(State.Aim.VisualAimCorrectionPitch, State.Aim.TargetVisualAimCorrectionPitch,
		SafeDeltaSeconds, InterpSpeed);

	State.Aim.VisualAimCorrectionYaw = FMath::FInterpTo(State.Aim.VisualAimCorrectionYaw,
		State.Aim.TargetVisualAimCorrectionYaw, SafeDeltaSeconds, InterpSpeed);
}

void UWeaponAimHandler::ClearAimSolution()
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	State.Aim.CurrentAimSolution = FWeaponAimSolution();
}

FVector UWeaponAimHandler::ResolveShotDirection(const FVector& MuzzleLocation, const FVector& AimTargetLocation) const
{

	return (AimTargetLocation - MuzzleLocation).GetSafeNormal();
}

bool UWeaponFireHandler::CanFireWeapon() const
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	const FWeaponRuntimeState& State = Component->RuntimeState;

	if (!Context.LinkedAnimationMesh
		|| !Component->HasEquippedWeapon()
		|| Context.EquippedWeaponInstance->IsBroken()
		|| State.Fire.bFireInProgress
		|| State.Fire.bContinuousAttackActive
		|| State.Carry.WeaponCarryState != EWeaponCarryState::Ready
		|| State.Shared.WeaponActionState != EWeaponActionState::Idle)
	{
		return false;
	}

	if (const UAbilitySystemComponent* AbilitySystem = Context.LinkedAbilitySystemComponent)
	{
		if (AbilitySystem->HasMatchingGameplayTag(AstraGameplayTags::State_Dead)
			|| AbilitySystem->HasMatchingGameplayTag(AstraGameplayTags::State_Stunned))
		{
			return false;
		}
	}

	if (Context.LinkedStatusModel && !Context.LinkedStatusModel->CanAttack())
	{
		return false;
	}

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData || WeaponData->AttackKind != EAttackKind::Ranged)
	{
		return false;
	}

	UAmmunitionDataAsset* AmmunitionData = Component->ResolveEquippedAmmunitionData();
	if (WeaponData->DeliveryType == EAttackDeliveryType::PhysicalProjectile && (!AmmunitionData || !AmmunitionData->ProjectileActorClass))
	{
		return false;
	}

	const int32 AmmoCost = GetInitialAttackAmmoCost();
	if (!WeaponData->bUsesAmmo || AmmoCost <= 0)
	{
		return true;
	}

	if (!Context.LinkedInventoryComponent || !Context.EquippedWeaponInstance->GetRegisteredAmmunitionData() || !Context.EquippedWeaponInstance->HasAmmo(AmmoCost))
	{
		return false;
	}

	const int32 ReservationQuantity = Context.LinkedInventoryComponent->GetReservationQuantity(Context.EquippedWeaponInstance->GetRuntimeItemId());
	checkf(ReservationQuantity == Context.EquippedWeaponInstance->GetCurrentAmmo(), TEXT("CanFireWeapon: Reservation과 WeaponInstance 등록 탄수가 일치하지 않습니다"));
	return ReservationQuantity >= AmmoCost;
}

bool UWeaponFireHandler::FireEquippedWeapon()
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!CanFireWeapon())
	{
		return false;
	}

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData || !WeaponData->FireMontage)
	{
		UE_LOG(
			LogTemp,
			Warning,
			TEXT("%s: Fire failed because FireMontage is missing."),
			*GetNameSafe(Component)
		);

		return false;
	}

	if (WeaponData->AttackKind == EAttackKind::Ranged && !State.Aim.CurrentAimSolution.bValid)
	{
		UE_LOG(LogTemp, Warning, TEXT("%s: Fire failed because the Player has not supplied an AimSolution."), *GetNameSafe(Component));
		return false;
	}

	UAnimInstance* AnimInstance = Context.LinkedAnimationMesh ? Context.LinkedAnimationMesh->GetAnimInstance() : nullptr;

	if (!AnimInstance)
	{
		UE_LOG(
			LogTemp,
			Warning,
			TEXT("%s: Fire failed because AnimInstance is missing."),
			*GetNameSafe(Component)
		);

		return false;
	}

	Component->StopAutoHolsterTimer();

	State.Fire.bFireInProgress = true;
	State.Fire.bFireCommitted = false;
	State.Fire.bLastFireSucceeded = false;
	State.Shared.WeaponActionState = EWeaponActionState::Firing;
	State.Aim.LockedFireAimSolution = State.Aim.CurrentAimSolution;
	State.Fire.ActiveAttackInstanceId = FGuid::NewGuid();

	State.Fire.ActiveFireAnimInstance = AnimInstance;
	State.Fire.ActiveFireMontage = WeaponData->FireMontage;

	AnimInstance->OnPlayMontageNotifyBegin.AddUniqueDynamic(
		Component,
		&UWeaponComponent::HandleFireMontageNotify
	);

	const float MontageLength = AnimInstance->Montage_Play(
		State.Fire.ActiveFireMontage,
		1.0f,
		EMontagePlayReturnType::MontageLength,
		0.0f,
		true
	);

	if (MontageLength <= 0.0f)
	{
		AnimInstance->OnPlayMontageNotifyBegin.RemoveDynamic(
			Component,
			&UWeaponComponent::HandleFireMontageNotify
		);

		State.Fire.bFireInProgress = false;
		State.Fire.bFireCommitted = false;
		State.Fire.bLastFireSucceeded = false;
		State.Shared.WeaponActionState = EWeaponActionState::Idle;
		State.Fire.ActiveFireAnimInstance = nullptr;
		State.Fire.ActiveFireMontage = nullptr;

		if (State.Carry.WeaponCarryState == EWeaponCarryState::Ready)
		{
			Component->NotifyWeaponActivity();
		}

		UE_LOG(
			LogTemp,
			Warning,
			TEXT("%s: Montage_Play failed for fire montage."),
			*GetNameSafe(Component)
		);

		return false;
	}

	FOnMontageEnded EndDelegate;
	EndDelegate.BindUObject(
		Component,
		&UWeaponComponent::HandleFireMontageEnded
	);
	AnimInstance->Montage_SetEndDelegate(
		EndDelegate,
		State.Fire.ActiveFireMontage
	);

	UE_LOG(
		LogTemp,
		Display,
		TEXT("%s: Weapon fire montage started."),
		*GetNameSafe(Component)
	);

	return true;
}

void UWeaponFireHandler::HandleFireMontageNotify(
	FName NotifyName,
	const FBranchingPointNotifyPayload& BranchingPointPayload)
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	const FAnimMontageInstance* MontageInstance = State.Fire.ActiveFireAnimInstance ? State.Fire.ActiveFireAnimInstance->GetActiveInstanceForMontage(State.Fire.ActiveFireMontage) : nullptr;
	if (!MontageInstance || MontageInstance->GetInstanceID() != BranchingPointPayload.MontageInstanceID)
	{
		return;
	}

	if (!State.Fire.bFireInProgress || State.Fire.bFireCommitted
		|| NotifyName != Component->FireCommitNotifyName)
	{
		return;
	}

	if (!Context.LinkedAnimationMesh || BranchingPointPayload.SkelMeshComponent != Context.LinkedAnimationMesh)
	{
		return;
	}

	const uint64 BindingRevision = Component->WeaponBindingRevision;
	const bool bShotCommitted = CommitFireShot();
	if (BindingRevision != Component->WeaponBindingRevision || !State.Fire.bFireInProgress)
	{
		return;
	}
	if (!bShotCommitted)
	{
		UE_LOG(
			LogTemp,
			Warning,
			TEXT("%s: Weapon_FireCommit failed to create a shot."),
			*GetNameSafe(Component)
		);

		CancelFireWeapon(0.05f);
		return;
	}

	State.Fire.bFireCommitted = true;

	UE_LOG(
		LogTemp,
		Display,
		TEXT("%s: Weapon fire committed."),
		*GetNameSafe(Component)
	);
}

bool UWeaponFireHandler::CommitFireShot()
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	const uint64 BindingRevision = Component->WeaponBindingRevision;
	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData || !State.Aim.LockedFireAimSolution.bValid)
	{
		return false;
	}

	const bool bContinuous = WeaponData->ShootType == EShootType::Continuous;
	const float DamageMultiplier = bContinuous ? WeaponData->InitialContinuousDamageMultiplier : 1.f;
	if (!Component->CommitAttack(GetInitialAttackAmmoCost(), DamageMultiplier, State.Fire.ActiveAttackInstanceId, EBeamEffectCommand::Start))
	{
		return false;
	}

	if (BindingRevision != Component->WeaponBindingRevision || !State.Fire.bFireInProgress)
	{
		return false;
	}

	if (bContinuous)
	{
		State.Fire.bContinuousAttackActive = true;
		State.Fire.ContinuousTickAccumulator = 0.f;
		State.Fire.ContinuousElapsedTime = 0.f;
		State.Fire.ContinuousTickIndex = 0;
	}

	Component->AccumulateRecoil();
	if (BindingRevision != Component->WeaponBindingRevision || !State.Fire.bFireInProgress)
	{
		return false;
	}

	if (UWorld* World = Context.World.Get())
	{
		World->GetTimerManager().SetTimer(State.Fire.FireCooldownTimerHandle, Component, &UWeaponComponent::FinishFireCooldown, FMath::Max(0.01f, WeaponData->AttackCooldown), false);
	}

	return true;
}

void UWeaponFireHandler::HandleFireMontageEnded(
	UAnimMontage* Montage,
	bool bInterrupted)
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!State.Fire.bFireInProgress || Montage != State.Fire.ActiveFireMontage)
	{
		return;
	}

	if (State.Fire.ActiveFireAnimInstance)
	{
		State.Fire.ActiveFireAnimInstance->OnPlayMontageNotifyBegin.RemoveDynamic(
			Component,
			&UWeaponComponent::HandleFireMontageNotify
		);
	}

	if (!bInterrupted && !State.Fire.bFireCommitted)
	{
		UE_LOG(
			LogTemp,
			Warning,
			TEXT(
				"%s: Fire montage ended without Weapon_FireCommit. "
				"No ammo was consumed."
			),
			*GetNameSafe(Component)
		);
	}

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!bInterrupted && State.Fire.bFireCommitted && WeaponData && WeaponData->ShootType == EShootType::Continuous)
	{
		State.Fire.bFireInProgress = false;
		State.Fire.bLastFireSucceeded = true;
		State.Fire.bFireCommitted = false;
		State.Fire.ActiveFireAnimInstance = nullptr;
		State.Fire.ActiveFireMontage = nullptr;
		if (State.Carry.WeaponCarryState == EWeaponCarryState::Ready)
		{
			Component->NotifyWeaponActivity();
		}
		return;
	}
	if (bInterrupted && State.Fire.bContinuousAttackActive)
	{
		StopContinuousAttack();
	}

	// Notify 이후 중단된 경우에도 이미 생성된 발사는 성공으로 유지합니다.
	FinishFireWeapon(State.Fire.bFireCommitted);
}

void UWeaponFireHandler::FinishFireWeapon(bool bSucceeded)
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!State.Fire.bFireInProgress)
	{
		return;
	}

	if (State.Fire.ActiveFireAnimInstance)
	{
		State.Fire.ActiveFireAnimInstance->OnPlayMontageNotifyBegin.RemoveDynamic(
			Component,
			&UWeaponComponent::HandleFireMontageNotify
		);
	}

	State.Fire.bFireInProgress = false;
	State.Fire.bLastFireSucceeded = bSucceeded;
	State.Fire.bFireCommitted = false;
	State.Fire.ActiveFireAnimInstance = nullptr;
	State.Fire.ActiveFireMontage = nullptr;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (bSucceeded && WeaponData && WeaponData->DeliveryType == EAttackDeliveryType::Beam && WeaponData->ShootType != EShootType::Continuous)
	{
		FBeamEffectRequest EffectRequest;
		EffectRequest.AttackInstanceId = State.Fire.ActiveAttackInstanceId;
		EffectRequest.Command = EBeamEffectCommand::Stop;
		EffectRequest.EffectSystem = WeaponData->BeamVFX;
		EffectRequest.AttachComponent = Component;
		EffectRequest.AttachSocketName = WeaponData->MuzzleSocketName;
		EffectRequest.EndLocationParameterName = WeaponData->BeamEndLocationParameterName;
		EffectRequest.StartLocation = Component->GetMuzzleTransform().GetLocation();
		EffectRequest.EndLocation = State.Fire.LastBeamEndLocation;
		Component->DispatchBeamEffect(EffectRequest);
	}

	if (!bSucceeded)
	{
		if (UWorld* World = Context.World.Get())
		{
			World->GetTimerManager().ClearTimer(State.Fire.FireCooldownTimerHandle);
		}

		if (State.Shared.WeaponActionState == EWeaponActionState::Firing)
		{
			State.Shared.WeaponActionState = EWeaponActionState::Idle;
		}
	}

	if (State.Carry.WeaponCarryState == EWeaponCarryState::Ready)
	{
		Component->NotifyWeaponActivity();
	}

	UE_LOG(
		LogTemp,
		Display,
		TEXT("%s: Weapon fire finished. Succeeded=%s"),
		*GetNameSafe(Component),
		bSucceeded ? TEXT("true") : TEXT("false")
	);

	if (!State.Fire.bContinuousAttackActive)
	{
		State.Fire.ActiveAttackInstanceId.Invalidate();
	}
	if (bSucceeded)
	{
		Component->BeginRecoilRecovery();
	}
	Component->OnWeaponFireFinished.Broadcast(bSucceeded);
}

void UWeaponFireHandler::CancelFireWeapon(float BlendOutTime)
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	StopContinuousAttack();

	if (!State.Fire.bFireInProgress)
	{
		return;
	}

	UAnimInstance* AnimInstance = State.Fire.ActiveFireAnimInstance;
	UAnimMontage* FireMontage = State.Fire.ActiveFireMontage;
	const bool bShotAlreadyCommitted = State.Fire.bFireCommitted;

	if (AnimInstance && FireMontage
		&& AnimInstance->Montage_IsPlaying(FireMontage))
	{
		AnimInstance->Montage_Stop(
			FMath::Max(0.0f, BlendOutTime),
			FireMontage
		);

		if (State.Fire.bFireInProgress)
		{
			FinishFireWeapon(bShotAlreadyCommitted);
		}

		return;
	}

	FinishFireWeapon(bShotAlreadyCommitted);
}

void UWeaponFireHandler::FinishFireCooldown()
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (State.Shared.WeaponActionState == EWeaponActionState::Firing && !State.Fire.bContinuousAttackActive)
	{
		State.Shared.WeaponActionState = EWeaponActionState::Idle;
	}
}

void UWeaponFireHandler::UpdateContinuousAttack(float DeltaSeconds)
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	const uint64 BindingRevision = Component->WeaponBindingRevision;
	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData || WeaponData->ShootType != EShootType::Continuous)
	{
		StopContinuousAttack();
		return;
	}

	State.Fire.ContinuousElapsedTime += DeltaSeconds;
	State.Fire.ContinuousTickAccumulator += DeltaSeconds;
	const float TickInterval = FMath::Max(0.01f, WeaponData->ContinuousTickInterval);
	while (State.Fire.bContinuousAttackActive && State.Fire.ContinuousTickAccumulator >= TickInterval)
	{
		State.Fire.ContinuousTickAccumulator -= TickInterval;
		++State.Fire.ContinuousTickIndex;
		if (State.Aim.CurrentAimSolution.bValid)
		{
			State.Aim.LockedFireAimSolution = State.Aim.CurrentAimSolution;
		}
		if (!Component->CommitAttack(GetContinuousAmmoCost(State.Fire.ContinuousTickIndex), GetContinuousDamageMultiplier(State.Fire.ContinuousTickIndex), State.Fire.ActiveAttackInstanceId, EBeamEffectCommand::Update))
		{
			if (BindingRevision != Component->WeaponBindingRevision)
			{
				return;
			}
			StopContinuousAttack();
		}
		else
		{
			if (BindingRevision != Component->WeaponBindingRevision || !State.Fire.bContinuousAttackActive)
			{
				return;
			}
			Component->AccumulateRecoil();
		}
	}
}

void UWeaponFireHandler::StopContinuousAttack()
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!State.Fire.bContinuousAttackActive)
	{
		return;
	}

	const bool bShouldBroadcastFinished = !State.Fire.bFireInProgress;
	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (WeaponData && WeaponData->DeliveryType == EAttackDeliveryType::Beam)
	{
		FBeamEffectRequest EffectRequest;
		EffectRequest.AttackInstanceId = State.Fire.ActiveAttackInstanceId;
		EffectRequest.Command = EBeamEffectCommand::Stop;
		EffectRequest.EffectSystem = WeaponData->BeamVFX;
		EffectRequest.AttachComponent = Component;
		EffectRequest.AttachSocketName = WeaponData->MuzzleSocketName;
		EffectRequest.EndLocationParameterName = WeaponData->BeamEndLocationParameterName;
		EffectRequest.StartLocation = Component->GetMuzzleTransform().GetLocation();
		EffectRequest.EndLocation = State.Fire.LastBeamEndLocation;
		Component->DispatchBeamEffect(EffectRequest);
	}

	State.Fire.bContinuousAttackActive = false;
	State.Fire.ContinuousTickAccumulator = 0.f;
	State.Fire.ContinuousElapsedTime = 0.f;
	State.Fire.ContinuousTickIndex = 0;
	State.Fire.ActiveAttackInstanceId.Invalidate();
	if (State.Shared.WeaponActionState == EWeaponActionState::Firing)
	{
		State.Shared.WeaponActionState = EWeaponActionState::Idle;
	}
	if (bShouldBroadcastFinished)
	{
		State.Fire.bLastFireSucceeded = true;
		Component->BeginRecoilRecovery();
		Component->OnWeaponFireFinished.Broadcast(true);
	}
}

int32 UWeaponFireHandler::GetInitialAttackAmmoCost() const
{
	UWeaponComponent* Component = GetWeaponComponent();

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData || !WeaponData->bUsesAmmo)
	{
		return 0;
	}
	return WeaponData->ShootType == EShootType::Continuous ? FMath::Max(0, WeaponData->InitialContinuousAmmoCost) : FMath::Max(1, WeaponData->AmmoPerShot);
}

int32 UWeaponFireHandler::GetContinuousAmmoCost(int32 TickIndex) const
{
	UWeaponComponent* Component = GetWeaponComponent();

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	return WeaponData ? FMath::Max(0, WeaponData->InitialContinuousAmmoCost + WeaponData->ContinuousAmmoCostIncreasePerTick * TickIndex) : 0;
}

float UWeaponFireHandler::GetContinuousDamageMultiplier(int32 TickIndex) const
{
	UWeaponComponent* Component = GetWeaponComponent();

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData)
	{
		return 0.f;
	}

	float DamageMultiplierIncrease = WeaponData->ContinuousDamageMultiplierIncreasePerTick * TickIndex;
	if (WeaponData->MaxDamageMultiplierIncreasePerTick > 0.f)
	{
		DamageMultiplierIncrease = FMath::Min(DamageMultiplierIncrease, WeaponData->MaxDamageMultiplierIncreasePerTick);
	}
	return FMath::Max(0.f, WeaponData->InitialContinuousDamageMultiplier + DamageMultiplierIncrease);
}

int32 UWeaponReloadHandler::CommitEquippedWeaponReload()
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;

	const TStrongObjectPtr<UWeaponInstance> Weapon(Context.EquippedWeaponInstance.Get());
	const TStrongObjectPtr<UInventoryComponent> Inventory(Context.LinkedInventoryComponent.Get());
	checkf(Weapon.Get() != nullptr, TEXT("CommitEquippedWeaponReload: WeaponInstance가 없습니다"));
	checkf(Inventory.Get() != nullptr, TEXT("CommitEquippedWeaponReload: InventoryComponent가 없습니다"));

	UAmmunitionDataAsset* AmmunitionData = ResolveReloadAmmunition();
	if (!AmmunitionData)
	{
		return 0;
	}

	const int32 CurrentAmmo = Weapon.Get()->GetCurrentAmmo();
	const int32 AvailableAmmo = Inventory.Get()->GetAvailableAmmunitionQuantity(AmmunitionData);
	const int32 AddedAmmo = FMath::Min(Weapon.Get()->GetMaxLoadedAmmo() - CurrentAmmo, AvailableAmmo);
	if (AddedAmmo <= 0)
	{
		return 0;
	}

	const int32 NewRegisteredAmmo = CurrentAmmo + AddedAmmo;
	if (!Inventory.Get()->TrySetAmmunitionReservation(Weapon.Get()->GetRuntimeItemId(), AmmunitionData, NewRegisteredAmmo))
	{
		return 0;
	}

	Weapon.Get()->CommitReload(AmmunitionData, NewRegisteredAmmo);
	return AddedAmmo;
}

bool UWeaponReloadHandler::CanReloadWeapon() const
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	const FWeaponRuntimeState& State = Component->RuntimeState;

	if (State.Reload.bReloadInProgress || State.Fire.bFireInProgress || State.Fire.bContinuousAttackActive)
	{
		return false;
	}

	if (!Context.LinkedAnimationMesh || !Context.EquippedWeaponInstance || Context.EquippedWeaponInstance->IsBroken())
	{
		return false;
	}

	if (State.Carry.WeaponCarryState == EWeaponCarryState::Drawing || State.Carry.WeaponCarryState == EWeaponCarryState::Holstering)
	{
		return false;
	}

	if (State.Shared.WeaponActionState != EWeaponActionState::Idle)
	{
		return false;
	}


	if (!Context.EquippedWeaponInstance->CanReload())
	{
		return false;
	}

	if (!Context.LinkedInventoryComponent || !ResolveReloadAmmunition())
	{
		return false;
	}

	if (const UAbilitySystemComponent* AbilitySystem = Context.LinkedAbilitySystemComponent)
	{
		FGameplayTagContainer BlockedTags;
		BlockedTags.AddTag(AstraGameplayTags::State_Dead);
		BlockedTags.AddTag(AstraGameplayTags::State_Stunned);
		BlockedTags.AddTag(AstraGameplayTags::State_Movement_Airborne);
		BlockedTags.AddTag(AstraGameplayTags::State_Movement_Traversal);
		BlockedTags.AddTag(AstraGameplayTags::State_Combat_Attacking);
		BlockedTags.AddTag(AstraGameplayTags::State_Weapon_Blocked);

		if (AbilitySystem->HasAnyMatchingGameplayTags(BlockedTags))
		{
			return false;
		}
	}

	// Sprint는 여기서 차단하지 않습니다.
	// 재장전 진입 이벤트가 Sprint를 취소하고 Walk로 전환합니다.
	return true;
}

bool UWeaponReloadHandler::ExecuteReloadWeapon()
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!CanReloadWeapon())
	{
		return false;
	}

	const UWeaponDataAsset* WeaponData =
		Context.EquippedWeaponInstance->GetWeaponData();

	if (!WeaponData || !WeaponData->ReloadMontage)
	{
		UE_LOG(
			LogTemp,
			Warning,
			TEXT("%s: Reload failed because ReloadMontage is missing."),
			*GetNameSafe(Component)
		);

		return false;
	}

	UAnimInstance* AnimInstance = Context.LinkedAnimationMesh ? Context.LinkedAnimationMesh->GetAnimInstance() : nullptr;

	if (!AnimInstance)
	{
		UE_LOG(
			LogTemp,
			Warning,
			TEXT("%s: Reload failed because AnimInstance is missing."),
			*GetNameSafe(Component)
		);

		return false;
	}

	Component->StopAutoHolsterTimer();

	State.Reload.bReloadInProgress = true;
	State.Reload.bReloadCommitted = false;
	State.Reload.bLastReloadSucceeded = false;

	State.Shared.WeaponActionState = EWeaponActionState::Reloading;

	State.Reload.ActiveReloadAnimInstance = AnimInstance;
	State.Reload.ActiveReloadMontage = WeaponData->ReloadMontage;

	AnimInstance->OnPlayMontageNotifyBegin.AddUniqueDynamic(Component, &UWeaponComponent::HandleReloadMontageNotify);

	const float MontageLength = AnimInstance->Montage_Play(State.Reload.ActiveReloadMontage, 1.0f, EMontagePlayReturnType::MontageLength, 0.0f, true);

	if (MontageLength <= 0.0f)
	{
		AnimInstance->OnPlayMontageNotifyBegin.RemoveDynamic(Component, &UWeaponComponent::HandleReloadMontageNotify);

		State.Reload.bReloadInProgress = false;
		State.Reload.bReloadCommitted = false;
		State.Reload.bLastReloadSucceeded = false;

		State.Shared.WeaponActionState = EWeaponActionState::Idle;

		State.Reload.ActiveReloadAnimInstance = nullptr;
		State.Reload.ActiveReloadMontage = nullptr;

		if (State.Carry.WeaponCarryState == EWeaponCarryState::Ready)
		{
			Component->NotifyWeaponActivity();
		}

		UE_LOG(
			LogTemp,
			Warning,
			TEXT("%s: Montage_Play failed for reload montage."),
			*GetNameSafe(Component)
		);

		return false;
	}

	FOnMontageEnded EndDelegate;

	EndDelegate.BindUObject(
		Component,
		&UWeaponComponent::HandleReloadMontageEnded
	);

	AnimInstance->Montage_SetEndDelegate(
		EndDelegate,
		State.Reload.ActiveReloadMontage
	);

	UE_LOG(
		LogTemp,
		Display,
		TEXT("%s: Weapon reload started."),
		*GetNameSafe(Component)
	);

	return true;
}

void UWeaponReloadHandler::HandleReloadMontageNotify(
	FName NotifyName,
	const FBranchingPointNotifyPayload& BranchingPointPayload)
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	const FAnimMontageInstance* MontageInstance = State.Reload.ActiveReloadAnimInstance ? State.Reload.ActiveReloadAnimInstance->GetActiveInstanceForMontage(State.Reload.ActiveReloadMontage) : nullptr;
	if (!MontageInstance || MontageInstance->GetInstanceID() != BranchingPointPayload.MontageInstanceID)
	{
		return;
	}

	if (!State.Reload.bReloadInProgress || State.Reload.bReloadCommitted)
	{
		return;
	}

	if (NotifyName != Component->ReloadCommitNotifyName)
	{
		return;
	}

	if (!Context.LinkedAnimationMesh || BranchingPointPayload.SkelMeshComponent != Context.LinkedAnimationMesh)
	{
		return;
	}

	const uint64 BindingRevision = Component->WeaponBindingRevision;
	const int32 AddedAmmo = CommitEquippedWeaponReload();
	if (BindingRevision != Component->WeaponBindingRevision || !State.Reload.bReloadInProgress)
	{
		return;
	}
	State.Reload.bReloadCommitted = AddedAmmo > 0;

	if (!State.Reload.bReloadCommitted)
	{
		UE_LOG(
			LogTemp,
			Warning,
			TEXT("%s: Weapon_ReloadCommit did not add ammo."),
			*GetNameSafe(Component)
		);
		return;
	}

	UE_LOG(
		LogTemp,
		Display,
		TEXT("%s: Reload ammo committed. AddedAmmo=%d"),
		*GetNameSafe(Component),
		AddedAmmo
	);
}

void UWeaponReloadHandler::HandleReloadMontageEnded(
	UAnimMontage* Montage,
	bool bInterrupted)
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!State.Reload.bReloadInProgress || Montage != State.Reload.ActiveReloadMontage)
	{
		return;
	}

	if (State.Reload.ActiveReloadAnimInstance)
	{
		State.Reload.ActiveReloadAnimInstance->OnPlayMontageNotifyBegin.RemoveDynamic(
			Component,
			&UWeaponComponent::HandleReloadMontageNotify
		);
	}

	const bool bSucceeded =
		!bInterrupted && State.Reload.bReloadCommitted;

	if (!bInterrupted && !State.Reload.bReloadCommitted)
	{
		UE_LOG(
			LogTemp,
			Warning,
			TEXT(
				"%s: Reload montage ended without "
				"'Weapon_ReloadCommit'. Ammo was not refilled."
			),
			*GetNameSafe(Component)
		);
	}

	FinishReloadWeapon(bSucceeded);
}

void UWeaponReloadHandler::FinishReloadWeapon(bool bSucceeded)
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!State.Reload.bReloadInProgress)
	{
		return;
	}

	if (State.Reload.ActiveReloadAnimInstance)
	{
		State.Reload.ActiveReloadAnimInstance->OnPlayMontageNotifyBegin.RemoveDynamic(
			Component,
			&UWeaponComponent::HandleReloadMontageNotify
		);
	}

	State.Reload.bReloadInProgress = false;
	State.Reload.bLastReloadSucceeded = bSucceeded;
	State.Reload.bReloadCommitted = false;

	if (State.Shared.WeaponActionState == EWeaponActionState::Reloading)
	{
		State.Shared.WeaponActionState = EWeaponActionState::Idle;
	}

	State.Reload.ActiveReloadAnimInstance = nullptr;
	State.Reload.ActiveReloadMontage = nullptr;

	if (State.Carry.WeaponCarryState == EWeaponCarryState::Ready)
	{
		Component->NotifyWeaponActivity();
	}

	UE_LOG(
		LogTemp,
		Display,
		TEXT("%s: Weapon reload finished. Succeeded=%s"),
		*GetNameSafe(Component),
		bSucceeded ? TEXT("true") : TEXT("false")
	);

	Component->OnWeaponReloadFinished.Broadcast(bSucceeded);
}

void UWeaponReloadHandler::CancelReloadWeapon(float BlendOutTime)
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!State.Reload.bReloadInProgress)
	{
		return;
	}

	UAnimInstance* AnimInstance = State.Reload.ActiveReloadAnimInstance;
	UAnimMontage* ReloadMontage = State.Reload.ActiveReloadMontage;

	if (AnimInstance && ReloadMontage &&
		AnimInstance->Montage_IsPlaying(ReloadMontage))
	{
		AnimInstance->Montage_Stop(
			FMath::Max(0.0f, BlendOutTime),
			ReloadMontage
		);

		/*
		 * 일반적으로 Montage_Stop이 EndDelegate를 실행합니다.
		 * 실행하지 않는 특수 상황도 처리하기 위한 안전장치입니다.
		 */
		if (State.Reload.bReloadInProgress)
		{
			FinishReloadWeapon(false);
		}

		return;
	}

	FinishReloadWeapon(false);
}

UAmmunitionDataAsset* UWeaponReloadHandler::ResolveReloadAmmunition() const
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;

	if (!Context.LinkedInventoryComponent || !Context.EquippedWeaponInstance || !Context.EquippedWeaponInstance->UsesAmmo())
	{
		return nullptr;
	}

	if (Context.EquippedWeaponInstance->GetCurrentAmmo() > 0)
	{
		UAmmunitionDataAsset* RegisteredAmmunition = Context.EquippedWeaponInstance->GetRegisteredAmmunitionData();
		return RegisteredAmmunition && Context.LinkedInventoryComponent->GetAvailableAmmunitionQuantity(RegisteredAmmunition) > 0 ? RegisteredAmmunition : nullptr;
	}

	const UWeaponDataAsset* WeaponData = Context.EquippedWeaponInstance->GetWeaponData();
	checkf(WeaponData != nullptr, TEXT("ResolveReloadAmmunition: WeaponData가 없습니다"));
	checkf(WeaponData->AmmoTypeTag.IsValid(), TEXT("ResolveReloadAmmunition: AmmoTypeTag가 유효하지 않습니다"));
	return Context.LinkedInventoryComponent->FindAvailableAmmunitionData(WeaponData->AmmoTypeTag, Context.EquippedWeaponInstance->GetLastUsedAmmunitionData());
}

void UWeaponRecoilHandler::AccumulateRecoil()
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData || !WeaponData->bUsesRecoil)
	{
		return;
	}

	State.Recoil.bRecoilRecoveryActive = false;
	State.Recoil.RecoilRecoveryElapsedTime = 0.f;
	const float PitchMin = FMath::Min(WeaponData->RecoilPitchPerShotMin, WeaponData->RecoilPitchPerShotMax);
	const float PitchMax = FMath::Max(WeaponData->RecoilPitchPerShotMin, WeaponData->RecoilPitchPerShotMax);
	const float YawMin = FMath::Min(WeaponData->RecoilYawPerShotMin, WeaponData->RecoilYawPerShotMax);
	const float YawMax = FMath::Max(WeaponData->RecoilYawPerShotMin, WeaponData->RecoilYawPerShotMax);
	const float MaxPitch = FMath::Max(0.f, WeaponData->RecoilPitchAccumulationLimit);
	const float MaxYaw = FMath::Max(0.f, WeaponData->RecoilYawAccumulationLimit);
	State.Recoil.CurrentRecoil.Pitch = FMath::Clamp(State.Recoil.CurrentRecoil.Pitch + FMath::FRandRange(PitchMin, PitchMax), -MaxPitch, MaxPitch);
	State.Recoil.CurrentRecoil.Yaw = FMath::Clamp(State.Recoil.CurrentRecoil.Yaw + FMath::FRandRange(YawMin, YawMax), -MaxYaw, MaxYaw);
	Component->BroadcastCurrentRecoil();
}

void UWeaponRecoilHandler::BeginRecoilRecovery()
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData || !WeaponData->bUsesRecoil || (FMath::IsNearlyZero(State.Recoil.CurrentRecoil.Pitch) && FMath::IsNearlyZero(State.Recoil.CurrentRecoil.Yaw)))
	{
		return;
	}

	State.Recoil.RecoilRecoveryStart = State.Recoil.CurrentRecoil;
	State.Recoil.RecoilRecoveryElapsedTime = 0.f;
	State.Recoil.bRecoilRecoveryActive = true;
	if (WeaponData->RecoilRecoveryDuration <= 0.f)
	{
		ResetRecoil();
	}
}

void UWeaponRecoilHandler::UpdateRecoilRecovery(float DeltaSeconds)
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	if (!State.Recoil.bRecoilRecoveryActive)
	{
		return;
	}

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData || !WeaponData->bUsesRecoil || WeaponData->RecoilRecoveryDuration <= 0.f)
	{
		ResetRecoil();
		return;
	}

	State.Recoil.RecoilRecoveryElapsedTime += FMath::Max(0.f, DeltaSeconds);
	const float RecoveryAlpha = FMath::Clamp(State.Recoil.RecoilRecoveryElapsedTime / WeaponData->RecoilRecoveryDuration, 0.f, 1.f);
	State.Recoil.CurrentRecoil.Pitch = FMath::Lerp(State.Recoil.RecoilRecoveryStart.Pitch, 0.f, RecoveryAlpha);
	State.Recoil.CurrentRecoil.Yaw = FMath::Lerp(State.Recoil.RecoilRecoveryStart.Yaw, 0.f, RecoveryAlpha);
	Component->BroadcastCurrentRecoil();
	if (RecoveryAlpha >= 1.f)
	{
		State.Recoil.bRecoilRecoveryActive = false;
		State.Recoil.RecoilRecoveryElapsedTime = 0.f;
		State.Recoil.RecoilRecoveryStart = FWeaponRecoilState();
	}
}

void UWeaponRecoilHandler::ResetRecoil()
{
	UWeaponComponent* Component = GetWeaponComponent();
	FWeaponRuntimeState& State = Component->RuntimeState;

	const bool bHadRecoil = !FMath::IsNearlyZero(State.Recoil.CurrentRecoil.Pitch) || !FMath::IsNearlyZero(State.Recoil.CurrentRecoil.Yaw);
	State.Recoil.CurrentRecoil = FWeaponRecoilState();
	State.Recoil.RecoilRecoveryStart = FWeaponRecoilState();
	State.Recoil.RecoilRecoveryElapsedTime = 0.f;
	State.Recoil.bRecoilRecoveryActive = false;
	if (bHadRecoil)
	{
		Component->BroadcastCurrentRecoil();
	}
}

float UWeaponRecoilHandler::GetNormalizedRecoilMagnitude() const
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeState& State = Component->RuntimeState;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData || !WeaponData->bUsesRecoil)
	{
		return 0.f;
	}

	const float MaxPitch = FMath::Max(0.f, WeaponData->RecoilPitchAccumulationLimit);
	const float MaxYaw = FMath::Max(0.f, WeaponData->RecoilYawAccumulationLimit);

	const float PitchAlpha = MaxPitch > KINDA_SMALL_NUMBER
		? FMath::Abs(State.Recoil.CurrentRecoil.Pitch) / MaxPitch
		: 0.f;

	const float YawAlpha = MaxYaw > KINDA_SMALL_NUMBER
		? FMath::Abs(State.Recoil.CurrentRecoil.Yaw) / MaxYaw
		: 0.f;

	return FMath::Clamp(FMath::Max(PitchAlpha, YawAlpha), 0.f, 1.f);
}

float UWeaponRecoilHandler::CalculateCurrentAccuracy(bool bAiming, float ExternalAccuracyModifier) const
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	const FWeaponRuntimeState& State = Component->RuntimeState;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData || !Context.EquippedWeaponInstance)
	{
		return 0.f;
	}

	const bool bMoving = State.Shared.CurrentNormalizedMoveSpeed > KINDA_SMALL_NUMBER;
	const float AimingBonus = bAiming ? WeaponData->AimingAccuracyBonus : 0.f;
	const float MovementPenalty = bMoving ? (bAiming ? WeaponData->MovingAimingAccuracyPenalty : WeaponData->MovingHipFireAccuracyPenalty) : 0.f;
	const float ContextAccuracy = WeaponData->BaseAccuracy + AimingBonus - MovementPenalty + ExternalAccuracyModifier;
	return FMath::Clamp(ContextAccuracy, 0.f, 100.f);
}

float UWeaponRecoilHandler::CalculateSpreadAngleDegrees(float Accuracy) const
{
	UWeaponComponent* Component = GetWeaponComponent();

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData)
	{
		return 0.f;
	}

	const float AccuracyRatio = FMath::Clamp(Accuracy / 100.f, 0.f, 1.f);
	return FMath::Max(0.f, WeaponData->MaxSpreadAngleDegrees) * (1.f - AccuracyRatio);
}

float UWeaponRecoilHandler::CalculateSpreadRadius(float Accuracy, float Distance) const
{

	const float SpreadAngleRadians = FMath::DegreesToRadians(CalculateSpreadAngleDegrees(Accuracy));
	return FMath::Tan(SpreadAngleRadians) * FMath::Max(0.f, Distance);
}

bool UWeaponAttackHandler::CommitAttack(int32 AmmoCost, float DamageMultiplier, const FGuid& AttackInstanceId, EBeamEffectCommand BeamCommand)
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData || !State.Aim.LockedFireAimSolution.bValid || !AttackInstanceId.IsValid())
	{
		return false;
	}

	const int32 ResolvedAmmoCost = WeaponData->bUsesAmmo ? FMath::Max(0, AmmoCost) : 0;
	if (WeaponData->bUsesAmmo && ResolvedAmmoCost > 0 && !Context.EquippedWeaponInstance->HasAmmo(ResolvedAmmoCost))
	{
		return false;
	}

	const FTransform MuzzleTransform = Component->GetMuzzleTransform();
	const FVector ShotDirection = Component->ResolveShotDirection(MuzzleTransform.GetLocation(), State.Aim.LockedFireAimSolution.AimTargetLocation);
	if (ShotDirection.IsNearlyZero())
	{
		return false;
	}

	FWeaponAttackCommitData CommitData;
	CommitData.AttackInstanceId = AttackInstanceId;
	CommitData.DeliveryType = WeaponData->DeliveryType;
	CommitData.bConsumesAmmo = WeaponData->bUsesAmmo && ResolvedAmmoCost > 0;
	CommitData.AmmoCost = ResolvedAmmoCost;
	const UAmmunitionDataAsset* AmmunitionData = Context.EquippedWeaponInstance->GetRegisteredAmmunitionData();
	CommitData.AmmoTypeTag = AmmunitionData ? AmmunitionData->AmmoTypeTag : WeaponData->AmmoTypeTag;
	CommitData.MuzzleTransform = FTransform(ShotDirection.Rotation(), MuzzleTransform.GetLocation());
	CommitData.AimTargetLocation = State.Aim.LockedFireAimSolution.AimTargetLocation;
	CommitData.AttackSpec = BuildAttackSpec(DamageMultiplier, ShotDirection);

	if (WeaponData->DeliveryType == EAttackDeliveryType::PhysicalProjectile)
	{
		CommitData.AmmunitionData = ResolveEquippedAmmunitionData();
	}

	UWorld* World = Context.World.Get();
	AActor* SourceActor = Context.SourceActor.Get();
	if (!World || !SourceActor)
	{
		return false;
	}
	if (CommitData.DeliveryType == EAttackDeliveryType::PhysicalProjectile
		&& (!CommitData.AmmunitionData || !CommitData.AmmunitionData->ProjectileActorClass || !World->GetSubsystem<UProjectileFactorySubsystem>()))
	{
		return false;
	}
	if (CommitData.DeliveryType != EAttackDeliveryType::PhysicalProjectile
		&& CommitData.DeliveryType != EAttackDeliveryType::Hitscan
		&& CommitData.DeliveryType != EAttackDeliveryType::Beam)
	{
		return false;
	}

	const uint64 BindingRevision = Component->WeaponBindingRevision;
	if (CommitData.bConsumesAmmo && !TryConsumeEquippedAmmo(CommitData.AmmoCost))
	{
		return false;
	}
	if (BindingRevision != Component->WeaponBindingRevision || Component->bResettingWeapon)
	{
		return false;
	}

	// 이미 소비한 탄약은 생성 실패 시에도 복구하지 않는 기존 정책을 유지
	if (!ExecuteAttackDelivery(CommitData, BeamCommand))
	{
		UE_LOG(LogTemp, Error, TEXT("Weapon attack delivery failed. AttackInstanceId=%s"), *CommitData.AttackInstanceId.ToString());
		return false;
	}
	Component->OnWeaponAttackCommitted.Broadcast(CommitData);
	return true;
}

bool UWeaponAttackHandler::ExecuteAttackDelivery(const FWeaponAttackCommitData& CommitData, EBeamEffectCommand BeamCommand)
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;
	FWeaponRuntimeState& State = Component->RuntimeState;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	UWorld* World = Context.World.Get();
	AActor* SourceActor = Context.SourceActor.Get();
	if (!WeaponData || !World || !SourceActor)
	{
		return false;
	}

	switch (CommitData.DeliveryType)
	{
	case EAttackDeliveryType::PhysicalProjectile:
	{
		UProjectileFactorySubsystem* Factory = World->GetSubsystem<UProjectileFactorySubsystem>();
		if (!Factory || !CommitData.AmmunitionData || !CommitData.AmmunitionData->ProjectileActorClass)
		{
			return false;
		}
		FProjectileLaunchRequest Request;
		Request.CommitData = CommitData;
		Request.AmmunitionData = CommitData.AmmunitionData;
		Request.SpawnTransform = CommitData.MuzzleTransform;
		APawn* InstigatorPawn = Cast<APawn>(SourceActor);
		return Factory->LaunchProjectile(Request, SourceActor, InstigatorPawn ? InstigatorPawn : SourceActor->GetInstigator());
	}
	case EAttackDeliveryType::Hitscan:
	{
		const FWeaponTraceResult TraceResult = TraceAttack(CommitData);
		ApplyTraceDamage(TraceResult);
		Component->OnWeaponTraceResolved.Broadcast(TraceResult);
		return true;
	}
	case EAttackDeliveryType::Beam:
	{
		const FWeaponTraceResult TraceResult = TraceAttack(CommitData);
		State.Fire.LastBeamEndLocation = TraceResult.bBlockingHit ? TraceResult.HitResult.ImpactPoint : TraceResult.TraceEnd;
		FBeamEffectRequest EffectRequest;
		EffectRequest.AttackInstanceId = CommitData.AttackInstanceId;
		EffectRequest.Command = BeamCommand;
		EffectRequest.EffectSystem = WeaponData->BeamVFX;
		EffectRequest.AttachComponent = Component;
		EffectRequest.AttachSocketName = WeaponData->MuzzleSocketName;
		EffectRequest.EndLocationParameterName = WeaponData->BeamEndLocationParameterName;
		EffectRequest.StartLocation = CommitData.MuzzleTransform.GetLocation();
		EffectRequest.EndLocation = State.Fire.LastBeamEndLocation;
		DispatchBeamEffect(EffectRequest);
		ApplyTraceDamage(TraceResult);
		Component->OnWeaponTraceResolved.Broadcast(TraceResult);
		return true;
	}
	default:
		return false;
	}
}

FWeaponTraceResult UWeaponAttackHandler::TraceAttack(const FWeaponAttackCommitData& CommitData) const
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;

	FWeaponTraceResult Result;
	Result.CommitData = CommitData;
	Result.TraceStart = CommitData.MuzzleTransform.GetLocation();
	Result.TraceEnd = CommitData.AimTargetLocation;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	UWorld* World = Context.World.Get();
	if (!WeaponData || !World)
	{
		return Result;
	}

	FCollisionQueryParams QueryParams(SCENE_QUERY_STAT(WeaponAttackTrace), true);
	QueryParams.AddIgnoredComponent(Component);
	if (Context.LinkedAnimationMesh)
	{
		QueryParams.AddIgnoredComponent(Context.LinkedAnimationMesh.Get());
	}
	if (Context.LinkedBodyCollisionComponent)
	{
		QueryParams.AddIgnoredComponent(Context.LinkedBodyCollisionComponent.Get());
	}

	Result.bBlockingHit = World->LineTraceSingleByChannel(Result.HitResult, Result.TraceStart, Result.TraceEnd, WeaponData->AttackTraceChannel, QueryParams);
	return Result;
}

FAttackSpec UWeaponAttackHandler::BuildAttackSpec(float DamageMultiplier, const FVector& ShotDirection) const
{
	UWeaponComponent* Component = GetWeaponComponent();

	FAttackSpec Result;
	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData)
	{
		return Result;
	}

	Result.bCritical = FMath::FRandRange(0.f, 100.f) < WeaponData->CriticalChancePercent;
	Result.FinalDamage = WeaponData->BaseDamage * FMath::Max(0.f, DamageMultiplier);
	if (Result.bCritical)
	{
		Result.FinalDamage *= WeaponData->CriticalDamageMultiplier;
	}

	Result.PierceChancePercent = FMath::Clamp(WeaponData->PierceProbabilityPercent, 0.f, 100.f);
	Result.EffectiveRange = WeaponData->EffectiveRange;
	Result.OutOfRangeDamageMultiplier = WeaponData->OutOfRangeDamageMultiplier;
	Result.ShotDirection = ShotDirection;
	return Result;
}

bool UWeaponAttackHandler::TryConsumeEquippedAmmo(const int32 Amount)
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;

	const TStrongObjectPtr<UWeaponInstance> Weapon(Context.EquippedWeaponInstance.Get());
	const TStrongObjectPtr<UInventoryComponent> Inventory(Context.LinkedInventoryComponent.Get());
	checkf(Amount > 0, TEXT("TryConsumeEquippedAmmo: Amount는 1 이상이어야 합니다"));

	if (!Weapon.Get())
	{
		return false;
	}

	if (!Weapon.Get()->UsesAmmo())
	{
		return true;
	}

	checkf(Inventory.Get() != nullptr, TEXT("TryConsumeEquippedAmmo: InventoryComponent가 없습니다"));
	checkf(Weapon.Get()->GetRegisteredAmmunitionData() != nullptr, TEXT("TryConsumeEquippedAmmo: 등록 탄약 Definition이 없습니다"));

	if (!Weapon.Get()->HasAmmo(Amount) || !Inventory.Get()->TryConsumeReservedAmmunition(Weapon.Get()->GetRuntimeItemId(), Amount))
	{
		return false;
	}

	const bool bConsumed = Weapon.Get()->TryConsumeAmmo(Amount);
	checkf(bConsumed, TEXT("TryConsumeEquippedAmmo: Reservation 소비 후 WeaponInstance 탄수 감소에 실패했습니다"));

	if (Weapon.Get()->GetCurrentAmmo() == 0)
	{
		const UWeaponDataAsset* WeaponData = Weapon.Get()->GetWeaponData();
		checkf(WeaponData != nullptr && WeaponData->AmmoTypeTag.IsValid(), TEXT("TryConsumeEquippedAmmo: Weapon AmmoTypeTag가 유효하지 않습니다"));
		if (UAmmunitionDataAsset* NextAmmunition = Inventory.Get()->FindAvailableAmmunitionData(WeaponData->AmmoTypeTag, Weapon.Get()->GetLastUsedAmmunitionData()))
		{
			const bool bAmmunitionRegistrationSucceeded = Weapon.Get()->TryRegisterAmmunition(NextAmmunition);
			checkf(bAmmunitionRegistrationSucceeded, TEXT("TryConsumeEquippedAmmo: 다음 탄약 Definition 등록에 실패했습니다"));
		}
	}

	return true;
}

UAmmunitionDataAsset* UWeaponAttackHandler::ResolveEquippedAmmunitionData() const
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;

	const UWeaponDataAsset* WeaponData = Component->GetEquippedWeaponData();
	if (!WeaponData)
	{
		return nullptr;
	}

	return Context.EquippedWeaponInstance ? Context.EquippedWeaponInstance->GetRegisteredAmmunitionData() : nullptr;
}

void UWeaponAttackHandler::ApplyTraceDamage(const FWeaponTraceResult& TraceResult) const
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;

	AActor* SourceActor = Context.SourceActor.Get();
	if (!SourceActor)
	{
		return;
	}
	const APawn* SourcePawn = Cast<APawn>(SourceActor);
	AController* SourceController = SourcePawn ? SourcePawn->GetController() : SourceActor->GetInstigatorController();
	AActor* TargetActor = TraceResult.bBlockingHit ? TraceResult.HitResult.GetActor() : nullptr;
	if (!TargetActor || TargetActor == SourceActor)
	{
		return;
	}

	float FinalDamage = TraceResult.CommitData.AttackSpec.FinalDamage;
	const float EffectiveRange = TraceResult.CommitData.AttackSpec.EffectiveRange;
	if (EffectiveRange > 0.f && FVector::Distance(TraceResult.TraceStart, TraceResult.HitResult.ImpactPoint) > EffectiveRange)
	{
		FinalDamage *= 1.f - FMath::Clamp(TraceResult.CommitData.AttackSpec.OutOfRangeDamageMultiplier, 0.f, 1.f);
	}

	if (UEntityHealthComponent* TargetHealth = TargetActor->FindComponentByClass<UEntityHealthComponent>())
	{
		TargetHealth->ApplyDamage(FinalDamage, SourceActor);
		return;
	}

	UGameplayStatics::ApplyDamage(TargetActor, FinalDamage, SourceController, SourceActor, UDamageType::StaticClass());
}

void UWeaponAttackHandler::DispatchBeamEffect(const FBeamEffectRequest& EffectRequest)
{
	UWeaponComponent* Component = GetWeaponComponent();
	const FWeaponRuntimeContext& Context = Component->RuntimeContext;

	UWorld* World = Context.World.Get();
	if (UCombatEffectSubsystem* Effects = World ? World->GetSubsystem<UCombatEffectSubsystem>() : nullptr)
	{
		Effects->HandleBeamEffectRequest(EffectRequest);
	}
	Component->OnBeamEffectRequested.Broadcast(EffectRequest);
}

bool UWeaponComponent::BindWeaponInstance(UWeaponInstance* WeaponInstance)
{
	if (bResettingWeapon || bWeaponRuntimeEnded)
	{
		return false;
	}
	EnsureHandlers();
	checkf(WeaponInstance != nullptr, TEXT("BindWeaponInstance: WeaponInstance가 nullptr입니다"));
	checkf(RuntimeContext.LinkedAnimationMesh != nullptr, TEXT("BindWeaponInstance: LinkedAnimationMesh가 없습니다"));

	if (RuntimeContext.EquippedWeaponInstance == WeaponInstance)
	{
		return true;
	}

	UWeaponDataAsset* WeaponData = WeaponInstance->GetWeaponData();
	if (!WeaponData || !WeaponData->WeaponStaticMesh)
	{
		UE_LOG(LogTemp, Error, TEXT("%s: WeaponStaticMesh is not assigned for %s"), *GetNameSafe(this), *GetNameSafe(WeaponData));
		return false;
	}

	if (!RuntimeContext.LinkedAnimationMesh->DoesSocketExist(WeaponData->AttachSocketName) || !RuntimeContext.LinkedAnimationMesh->DoesSocketExist(WeaponData->HolsterSocketName))
	{
		UE_LOG(LogTemp, Error, TEXT("%s: Required weapon socket is missing Hand='%s' Holster='%s'"), *GetNameSafe(this), *WeaponData->AttachSocketName.ToString(), *WeaponData->HolsterSocketName.ToString());
		return false;
	}

	TGuardValue<bool> ResetGuard(bResettingWeapon, true);
	++WeaponBindingRevision;
	RuntimeState.Carry.PendingWeaponAction = EPendingWeaponAction::None;
	CancelFireWeapon(0.f);
	CancelReloadWeapon(0.f);
	CancelDrawWeapon(0.f);
	CancelHolsterWeapon(0.f);
	StopAutoHolsterTimer();
	if (UWorld* World = GetWorld())
	{
		World->GetTimerManager().ClearTimer(RuntimeState.Fire.FireCooldownTimerHandle);
	}
	RuntimeState.Aim.LockedFireAimSolution = FWeaponAimSolution();
	RuntimeState.Aim.VisualAimCorrectionPitch = 0.f;
	RuntimeState.Aim.VisualAimCorrectionYaw = 0.f;
	RuntimeState.Aim.TargetVisualAimCorrectionPitch = 0.f;
	RuntimeState.Aim.TargetVisualAimCorrectionYaw = 0.f;
	ClearAimSolution();
	ResetRecoil();

	if (!AttachWeaponToCharacterSocket(WeaponData->HolsterSocketName, WeaponData->HolsteredMeshTransform))
	{
		return false;
	}

	if (RuntimeContext.EquippedWeaponInstance)
	{
		RuntimeContext.EquippedWeaponInstance->OnAmmoChanged.RemoveDynamic(this, &UWeaponComponent::HandleEquippedWeaponAmmoChanged);
		RuntimeContext.EquippedWeaponInstance->OnRegisteredAmmunitionChanged.RemoveDynamic(this, &UWeaponComponent::HandleRegisteredAmmunitionChanged);
	}

	RuntimeContext.EquippedWeaponInstance = WeaponInstance;
	if (RuntimeContext.EquippedWeaponInstance->UsesAmmo() && RuntimeContext.EquippedWeaponInstance->GetCurrentAmmo() == 0)
	{
		if (UAmmunitionDataAsset* AmmunitionData = ResolveReloadAmmunition())
		{
			const bool bAmmunitionRegistrationSucceeded = RuntimeContext.EquippedWeaponInstance->TryRegisterAmmunition(AmmunitionData);
			checkf(bAmmunitionRegistrationSucceeded, TEXT("BindWeaponInstance: 초기 탄약 Definition 등록에 실패했습니다"));
		}
	}
	RuntimeState.Shared.WeaponActionState = EWeaponActionState::Idle;
	RuntimeState.Carry.PendingWeaponAction = EPendingWeaponAction::None;
	RuntimeState.Carry.bWeaponInHand = false;
	SetWeaponCarryState(EWeaponCarryState::Holstered);

	SetStaticMesh(WeaponData->WeaponStaticMesh);
	SetVisibility(true, true);
	SetHiddenInGame(false);

	RuntimeContext.EquippedWeaponInstance->OnAmmoChanged.AddUniqueDynamic(this, &UWeaponComponent::HandleEquippedWeaponAmmoChanged);
	RuntimeContext.EquippedWeaponInstance->OnRegisteredAmmunitionChanged.AddUniqueDynamic(this, &UWeaponComponent::HandleRegisteredAmmunitionChanged);

	BroadcastAmmoState();
	OnWeaponEquipped.Broadcast(RuntimeContext.EquippedWeaponInstance);

	UE_LOG(LogTemp, Display, TEXT("%s: Bound weapon '%s' Ammo=%d/%d HolsterSocket='%s'"), *GetNameSafe(this), *WeaponData->ItemName.ToString(), GetCurrentAmmo(), GetMaxLoadedAmmo(), *WeaponData->HolsterSocketName.ToString());

	return true;
}

void UWeaponComponent::ClearWeaponInstance()
{
	if (bResettingWeapon)
	{
		return;
	}
	EnsureHandlers();
	TGuardValue<bool> ResetGuard(bResettingWeapon, true);
	++WeaponBindingRevision;
	RuntimeState.Carry.PendingWeaponAction = EPendingWeaponAction::None;
	const bool bHadWeapon = RuntimeContext.EquippedWeaponInstance != nullptr;

	CancelFireWeapon(0.f);
	StopContinuousAttack();
	CancelReloadWeapon(0.f);
	StopAutoHolsterTimer();
	CancelDrawWeapon(0.f);
	CancelHolsterWeapon(0.f);
	RuntimeState.Aim.LockedFireAimSolution = FWeaponAimSolution();
	RuntimeState.Aim.VisualAimCorrectionPitch = 0.f;
	RuntimeState.Aim.VisualAimCorrectionYaw = 0.f;
	RuntimeState.Aim.TargetVisualAimCorrectionPitch = 0.f;
	RuntimeState.Aim.TargetVisualAimCorrectionYaw = 0.f;
	ClearAimSolution();
	ResetRecoil();

	if (UWorld* World = GetWorld())
	{
		World->GetTimerManager().ClearTimer(RuntimeState.Fire.FireCooldownTimerHandle);
	}

	RuntimeState.Carry.PendingWeaponAction = EPendingWeaponAction::None;

	if (RuntimeContext.EquippedWeaponInstance)
	{
		RuntimeContext.EquippedWeaponInstance->OnAmmoChanged.RemoveDynamic(this, &UWeaponComponent::HandleEquippedWeaponAmmoChanged);
		RuntimeContext.EquippedWeaponInstance->OnRegisteredAmmunitionChanged.RemoveDynamic(this, &UWeaponComponent::HandleRegisteredAmmunitionChanged);
	}

	RuntimeContext.EquippedWeaponInstance = nullptr;
	RuntimeState.Shared.WeaponActionState = EWeaponActionState::Idle;
	RuntimeState.Carry.bWeaponInHand = false;
	SetWeaponCarryState(EWeaponCarryState::Holstered);

	SetStaticMesh(nullptr);
	SetRelativeTransform(FTransform::Identity);
	SetVisibility(false, true);
	SetHiddenInGame(true);

	if (bHadWeapon)
	{
		BroadcastAmmoState();
		OnWeaponUnequipped.Broadcast();
	}
}

bool UWeaponComponent::RequestWeaponFire()
{
	if (bResettingWeapon || bWeaponRuntimeEnded)
	{
		return false;
	}
	UE_LOG(LogTemp, Warning,
		TEXT("%s: RequestWeaponFire entered. HasWeapon=%s CarryState=%s"),
		*GetNameSafe(this),
		HasEquippedWeapon() ? TEXT("true") : TEXT("false"),
		*UEnum::GetValueAsString(RuntimeState.Carry.WeaponCarryState));

	if (!HasEquippedWeapon())
	{
		UE_LOG(LogTemp, Warning, TEXT("%s: RequestWeaponFire stopped because no weapon is equipped."), *GetNameSafe(this));
		return false;
	}
	if (RejectBrokenWeaponUse())
	{
		return false;
	}

	const bool bReloadInstead = ShouldReloadOnFireRequest();
	const EPendingWeaponAction RequestedAction = bReloadInstead ? EPendingWeaponAction::Reload : EPendingWeaponAction::Fire;

	if (RuntimeState.Carry.WeaponCarryState != EWeaponCarryState::Ready)
	{
		return RequestWeaponDraw(RequestedAction);
	}

	NotifyWeaponActivity();
	return BroadcastWeaponCommand(bReloadInstead ? EWeaponCommand::Reload : EWeaponCommand::Fire);
}

bool UWeaponComponent::RequestWeaponAimStarted()
{
	if (bResettingWeapon || bWeaponRuntimeEnded)
	{
		return false;
	}
	if (!HasEquippedWeapon())
	{
		return false;
	}
	if (RejectBrokenWeaponUse())
	{
		return false;
	}

	const bool bAlreadyAiming = RuntimeContext.LinkedAbilitySystemComponent
		&& RuntimeContext.LinkedAbilitySystemComponent->HasMatchingGameplayTag(AstraGameplayTags::State_Combat_Aiming);

	if (!bAlreadyAiming && RuntimeState.Carry.WeaponCarryState != EWeaponCarryState::Ready)
	{
		return RequestWeaponDraw(EPendingWeaponAction::Aim);
	}

	if (RuntimeState.Carry.WeaponCarryState == EWeaponCarryState::Ready)
	{
		NotifyWeaponActivity();
	}

	return BroadcastWeaponCommand(EWeaponCommand::AimStarted);
}

	if (bResettingWeapon || bWeaponRuntimeEnded)
	{
		return false;

	{

	}

	if (RuntimeState.Carry.PendingWeaponAction == EPendingWeaponAction::Aim)
	{
		RuntimeState.Carry.PendingWeaponAction = EPendingWeaponAction::None;
	}

	return BroadcastWeaponCommand(EWeaponCommand::AimCompleted);
}

bool UWeaponComponent::RequestWeaponDraw(EPendingWeaponAction ActionAfterDraw)
{
	if (bResettingWeapon || bWeaponRuntimeEnded)
	{
		return false;
	}
	UE_LOG(LogTemp, Warning,
		TEXT("%s: RequestWeaponDraw entered. HasWeapon=%s CarryState=%s ActionAfterDraw=%s"),
		*GetNameSafe(this),
		HasEquippedWeapon() ? TEXT("true") : TEXT("false"),
		*UEnum::GetValueAsString(RuntimeState.Carry.WeaponCarryState),
		*UEnum::GetValueAsString(ActionAfterDraw));

	if (!HasEquippedWeapon())
	{
		return false;
	}

	if (RuntimeState.Carry.WeaponCarryState == EWeaponCarryState::Ready)
	{
		NotifyWeaponActivity();
		return true;
	}

	if (ActionAfterDraw != EPendingWeaponAction::None)
	{
		RuntimeState.Carry.PendingWeaponAction = ActionAfterDraw;
	}

	if (RuntimeState.Carry.WeaponCarryState == EWeaponCarryState::Drawing || RuntimeState.Carry.WeaponCarryState == EWeaponCarryState::Holstering)
	{
		return true;
	}

	return BroadcastWeaponCommand(EWeaponCommand::Draw);
}

bool UWeaponComponent::RequestWeaponHolster()
{
	if (bResettingWeapon || bWeaponRuntimeEnded)
	{
		return false;
	}
	RuntimeState.Carry.PendingWeaponAction = EPendingWeaponAction::None;
	CancelFireWeapon(0.1f);
	CancelReloadWeapon(0.1f);
	StopAutoHolsterTimer();

	if (RuntimeState.Carry.WeaponCarryState == EWeaponCarryState::Holstered
		|| RuntimeState.Carry.WeaponCarryState == EWeaponCarryState::Holstering)
	{
		return true;
	}

	if (RuntimeState.Carry.WeaponCarryState == EWeaponCarryState::Drawing)
	{
		ForceWeaponHolster();
		return true;
	}

	return BroadcastWeaponCommand(EWeaponCommand::Holster);
}

void APlayerCharacter::InitClass()
{
	if (bInitializationCompleted)
	{
		return;
	}

	ApplyDefaultInputMapping();

	if (!PlayerDataAsset)
	{
		UE_LOG(LogTemp, Error, TEXT("%s: PlayerDataAsset is not assigned"), *GetNameSafe(this));
		return;
	}

	if (!StatusModel || !LocomotionComponent)
	{
		UE_LOG(LogTemp, Error, TEXT("%s: Locomotion dependencies are missing"), *GetNameSafe(this));
		return;
	}

	StatusModel->InitializeStats(PlayerDataAsset);
	LocomotionComponent->InitializeLocomotion(StatusModel, this);

	if (!InitializeAbilitySystemFromPlayerState())
	{
		UE_LOG(LogTemp, Error, TEXT("%s: AstraPlayerState/ASC initialization failed"), *GetNameSafe(this));
		return;
	}

	if (!WeaponComponent)
	{
		WeaponComponent = FindComponentByClass<UWeaponComponent>();
	}

	checkf(WeaponComponent != nullptr, TEXT("InitClass: WeaponComponent가 없습니다"));

	WeaponComponent->InitializeWeapon(StatusModel, GetMesh(), GetCapsuleComponent(), AbilitySystemComponent);
	AAstraPlayerState* InventoryPlayerState = GetPlayerState<AAstraPlayerState>();
	checkf(InventoryPlayerState != nullptr && InventoryPlayerState->InventoryComponent != nullptr, TEXT("InitClass: PlayerState InventoryComponent가 없습니다"));
	WeaponComponent->SetInventoryComponent(InventoryPlayerState->InventoryComponent);
	WeaponComponent->OnWeaponCommandRequested.AddUniqueDynamic(this, &APlayerCharacter::HandleWeaponCommandRequested);
	WeaponComponent->OnPendingWeaponActionReady.AddUniqueDynamic(this, &APlayerCharacter::HandlePendingWeaponActionReady);
	WeaponComponent->OnWeaponCarryStateChanged.AddUniqueDynamic(this, &APlayerCharacter::HandleWeaponCarryStateChanged);
	WeaponComponent->OnAmmoChanged.AddUniqueDynamic(this, &APlayerCharacter::HandleWeaponAmmoChanged);
	WeaponComponent->OnWeaponEquipped.AddUniqueDynamic(this, &APlayerCharacter::HandleWeaponEquipped);
	WeaponComponent->OnWeaponUnequipped.AddUniqueDynamic(this, &APlayerCharacter::HandleWeaponUnequipped);
	WeaponComponent->OnWeaponFireFinished.AddUniqueDynamic(this, &APlayerCharacter::HandleWeaponFireFinished);
	WeaponComponent->OnWeaponReloadFinished.AddUniqueDynamic(this, &APlayerCharacter::HandleWeaponReloadFinished);
	WeaponComponent->OnWeaponRecoilChanged.AddUniqueDynamic(this, &APlayerCharacter::HandleWeaponRecoilChanged);
	WeaponComponent->OnWeaponUseFailed.AddUniqueDynamic(this, &APlayerCharacter::HandleWeaponUseFailed);

	RefreshPlayerEquipmentBinding();
	ApplyWeaponRotationMode(WeaponComponent->GetWeaponCarryState() != EWeaponCarryState::Holstered);

	if (PlayerContext)
	{
		PlayerContext->Initialize(StatusModel, LocomotionComponent, WeaponComponent, AbilitySystemComponent, HealthComponent, StateMediatorComponent, StateMediatorComponent ? StateMediatorComponent->GetEventBus() : nullptr);
	}

	if (BaseStateTreeComponent)
	{
		BaseStateTreeComponent->StartLogic();
	}

	if (WeaponCarryStateTreeComponent)
	{
		WeaponCarryStateTreeComponent->StartLogic();
	}

	if (AAstraPlayerState* AstraPlayerState = GetPlayerState<AAstraPlayerState>())
	{
		for (const TSubclassOf<UGameplayAbility> AbilityClass : StartupAbilityClasses)
		{
			if (!AbilityClass)
			{
				UE_LOG(LogTemp, Warning, TEXT("%s: StartupAbilityClasses contains an empty entry"), *GetNameSafe(this));
				continue;
			}

			AstraPlayerState->GrantAbility(AbilityClass, 1);
		}
	}

	bInitializationCompleted = true;
	UE_LOG(LogTemp, Log, TEXT("%s initialized with PlayerState-owned GAS, Inventory and Equipment"), *GetNameSafe(this));
}
bool APlayerCharacter::InitializeAbilitySystemFromPlayerState()
{
	AAstraPlayerState* AstraPlayerState = GetPlayerState<AAstraPlayerState>();
	if (!AstraPlayerState)
	{
		AbilitySystemComponent = nullptr;
		return false;
	}

	AstraPlayerState->InitializeAbilityActor(this);
	AstraPlayerState->InitializeAttributes(PlayerDataAsset, Cast<UPlayerDataAsset>(PlayerDataAsset));
	AbilitySystemComponent = AstraPlayerState->GetAbilitySystemComponent();
	if (WeaponComponent)
	{
		WeaponComponent->SetAbilitySystemComponent(AbilitySystemComponent);
		WeaponComponent->SetInventoryComponent(AstraPlayerState->InventoryComponent);
	}
	const bool bHealthReady = HealthComponent && HealthComponent->InitializeWithAbilitySystem(AbilitySystemComponent);
	const bool bMediatorReady = StateMediatorComponent && StateMediatorComponent->InitializeMediator(
		BaseStateTreeComponent, WeaponCarryStateTreeComponent, HealthComponent, AbilitySystemComponent);
	
	return AbilitySystemComponent && bHealthReady && bMediatorReady;
}

void APlayerCharacter::RefreshPlayerEquipmentBinding()
{
	AAstraPlayerState* AstraPlayerState = GetPlayerState<AAstraPlayerState>();
	checkf(AstraPlayerState != nullptr, TEXT("RefreshPlayerEquipmentBinding: AstraPlayerState가 없습니다"));
	checkf(AstraPlayerState->GearComponent != nullptr, TEXT("RefreshPlayerEquipmentBinding: EquipmentComponent가 없습니다"));

	UGearComponent* NewEquipmentComponent = AstraPlayerState->GearComponent;

	if (LinkedGearComponent != NewEquipmentComponent)
	{
		ClearPlayerEquipmentBinding();
		LinkedGearComponent = NewEquipmentComponent;
		LinkedGearComponent->OnActiveWeaponChanged.AddUniqueDynamic(this, &APlayerCharacter::HandleActiveWeaponChanged);
	}

	if (!LinkedGearComponent->GetActiveWeapon() && StartingWeaponData && !InitializeStartingEquipment())
	{
		UE_LOG(LogTemp, Warning, TEXT("%s: Starting weapon could not be added to PlayerState equipment"), *GetNameSafe(this));
	}

	HandleActiveWeaponChanged(LinkedGearComponent->GetActiveWeapon());
}

void APlayerCharacter::ClearPlayerEquipmentBinding()
{
	if (!LinkedGearComponent)
	{
		return;
	}

	LinkedGearComponent->OnActiveWeaponChanged.RemoveDynamic(this, &APlayerCharacter::HandleActiveWeaponChanged);
	LinkedGearComponent = nullptr;
}

bool APlayerCharacter::InitializeStartingEquipment()
{
	checkf(LinkedGearComponent != nullptr, TEXT("InitializeStartingEquipment: EquipmentComponent가 없습니다"));
	checkf(StartingWeaponData != nullptr, TEXT("InitializeStartingEquipment: StartingWeaponData가 없습니다"));

	if (LinkedGearComponent->GetActiveWeapon())
	{
		return true;
	}

	if (Cast<UWeaponInstance>(LinkedGearComponent->GetEquipmentAtSlot(EEquipmentSlot::Weapon1)))
	{
		return LinkedGearComponent->TrySetActiveWeaponSlot(EEquipmentSlot::Weapon1);
	}

	if (Cast<UWeaponInstance>(LinkedGearComponent->GetEquipmentAtSlot(EEquipmentSlot::Weapon2)))
	{
		return LinkedGearComponent->TrySetActiveWeaponSlot(EEquipmentSlot::Weapon2);
	}

	UWorld* World = GetWorld();
	checkf(World != nullptr, TEXT("InitializeStartingEquipment: World가 없습니다"));

	UItemInstanceFactorySubsystem* ItemInstanceFactory = World->GetSubsystem<UItemInstanceFactorySubsystem>();
	checkf(ItemInstanceFactory != nullptr, TEXT("InitializeStartingEquipment: ItemInstanceFactorySubsystem이 없습니다"));

	UWeaponInstance* StartingWeaponInstance = ItemInstanceFactory->CreateWeaponInstance(StartingWeaponData);
	if (!StartingWeaponInstance)
	{
		return false;
	}

	return LinkedGearComponent->TryEquip(StartingWeaponInstance, EEquipmentSlot::Weapon1);
}

void APlayerCharacter::HandleActiveWeaponChanged(UWeaponInstance* WeaponInstance)
{
	checkf(WeaponComponent != nullptr, TEXT("HandleActiveWeaponChanged: WeaponComponent가 없습니다"));

	if (!WeaponInstance)
	{
		WeaponComponent->ClearWeaponInstance();
		return;
	}

	if (WeaponComponent->GetEquippedWeaponInstance() == WeaponInstance)
	{
		return;
	}

	if (!WeaponComponent->BindWeaponInstance(WeaponInstance))
	{
		UE_LOG(LogTemp, Error, TEXT("%s: Active WeaponInstance '%s' could not be bound to WeaponComponent"), *GetNameSafe(this), *GetNameSafe(WeaponInstance));
	}
}
void APlayerCharacter::RefreshPlayerCameraManagerContext()
{
	if (!PlayerContext)
	{
		return;
	}

	AAstraPlayerCameraManager* AstraCameraManager = nullptr;
	if (const APlayerController* PlayerController = Cast<APlayerController>(GetController()))
	{
		if (PlayerController->IsLocalController())
		{
			AstraCameraManager = Cast<AAstraPlayerCameraManager>(PlayerController->PlayerCameraManager);
		}
	}

	if (PlayerContext->PlayerCameraManager && PlayerContext->PlayerCameraManager != AstraCameraManager)
	{
		PlayerContext->PlayerCameraManager->SetWeaponRecoilOffset(0.f, 0.f);
	}
	PlayerContext->SetPlayerCameraManager(AstraCameraManager);
	if (AstraCameraManager && WeaponComponent)
	{
		const FWeaponRecoilState RecoilState = WeaponComponent->GetCurrentRecoil();
		AstraCameraManager->SetWeaponRecoilOffset(RecoilState.Pitch, RecoilState.Yaw);
	}
}

bool APlayerCharacter::HandleControlCommand_Implementation(EEntityControlCommand Command)
{
	if (!WeaponComponent)
	{
		return false;
	}

	switch (Command)
	{
	case EEntityControlCommand::PrimaryStarted:
		return WeaponComponent->RequestWeaponFire();

	case EEntityControlCommand::PrimaryCompleted:
		return WeaponComponent->RequestWeaponFireCompleted();

	case EEntityControlCommand::AimStarted:
		return WeaponComponent->RequestWeaponAimStarted();

	case EEntityControlCommand::AimCompleted:
		return WeaponComponent->RequestWeaponAimCompleted();

	case EEntityControlCommand::Reload:
		return WeaponComponent->RequestWeaponReload();

	default:
		return false;
	}
}
void APlayerCharacter::HandlePendingWeaponActionReady(EPendingWeaponAction PendingAction)
{
	switch (PendingAction)
	{
	case EPendingWeaponAction::Fire:
		HandleWeaponCommandRequested(EWeaponCommand::Fire);

	case EPendingWeaponAction::Reload:
		HandleWeaponCommandRequested(EWeaponCommand::Reload);
		break;

	case EPendingWeaponAction::Aim:
		HandleWeaponCommandRequested(EWeaponCommand::AimStarted);

	default:
		break;
	}
}

FPlayerCrosshairState APlayerCharacter::GetCurrentPlayerCrosshairState() const
{
	FPlayerCrosshairState CrosshairState;

	if (!WeaponComponent)
	{
		return CrosshairState;
	}

	const UWeaponDataAsset* WeaponData = WeaponComponent->GetEquippedWeaponData();

	CrosshairState.bVisible = WeaponComponent->IsWeaponInHand()
		&& WeaponComponent->GetWeaponCarryState() == EWeaponCarryState::Ready;

	CrosshairState.WeaponType = WeaponData ? WeaponData->WeaponType : EWeaponType::None;

	CrosshairState.bAiming = AbilitySystemComponent 
		&& AbilitySystemComponent ->HasMatchingGameplayTag(AstraGameplayTags::State_Combat_Aiming);

	if (StatusModel)
	{
		CrosshairState.bAirborne = !StatusModel->IsGrounded();
	}
	else if (const UCharacterMovementComponent* MovementComponent = GetCharacterMovement())
	{
		CrosshairState.bAirborne = MovementComponent->IsFalling();
	}

	CrosshairState.NormalizedMoveSpeed =
		WeaponComponent->GetCurrentNormalizedMoveSpeed();

	CrosshairState.RecoilMagnitude =
		WeaponComponent->GetNormalizedRecoilMagnitude();

	return CrosshairState;
}
void APlayerCharacter::PublishPlayerCrosshairState()
{
	const FPlayerCrosshairState CrosshairState = GetCurrentPlayerCrosshairState();

	if (CrosshairState.bVisible)
	{
		bCrosshairWasVisible = true;
		OnPlayerCrosshairChanged.Broadcast(CrosshairState);
		return;
	}

	if (!bCrosshairWasVisible)
	{
		return;
	}

	bCrosshairWasVisible = false;
	OnPlayerCrosshairChanged.Broadcast(CrosshairState);
}

void APlayerCharacter::ApplyWeaponRotationMode(bool bWeaponActive)
{
	UCharacterMovementComponent* Movement = GetCharacterMovement();
	if (!Movement)
	{
		return;
	}

	bUseControllerRotationYaw = bWeaponActive;
	Movement->bOrientRotationToMovement = !bWeaponActive;
	Movement->bUseControllerDesiredRotation = false;

	if (bWeaponActive)
	{
		Movement->RotationRate.Yaw = WeaponFacingRotationRateYaw;
	}
	else if (bUnarmedRotationRateCached)
	{
		Movement->RotationRate = UnarmedRotationRate;
	}
}

			Subsystem->RemoveMappingContext(DefaultMappingContext);
			Subsystem->AddMappingContext(DefaultMappingContext, 0);
		}
	}
}
